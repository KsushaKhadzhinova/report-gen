# --- Базовые типы нотации A^(t), B^(t), E^(t), C^(t), X^(t) ---

Agent = int

Assignment = Dict[Edge, int]                    # X^(t): (i,j) -> {0,1}

# --- Утилиты для работы с назначением ---

def objective_static(C: CostMatrix, X: Assignment) -> float:
    """
    Статическая целевая функция для шага t:
    F^(t)(X^(t)) = sum_{i in A^(t)} sum_{j in B^(t)} c_ij^(t) * x_ij^(t)  (формула (1)).
    """
    return sum(C[e] * X.get(e, 0) for e in C.keys())

def phi_changed_edges(X_prev: Assignment, X_new: Assignment) -> int:
    """
    Φ(X^(t-1), X^(t)) — количество дуг, у которых изменилось x_ij (0→1 или 1→0).[file:79]
    """
    changed = 0
    all_edges = set(X_prev.keys()) | set(X_new.keys())
    for e in all_edges:
        if X_prev.get(e, 0) != X_new.get(e, 0):
            changed += 1
    return changed

def objective_dynamic(
    C_t: CostMatrix,
    X_prev: Assignment,
    X_t: Assignment,
    lambda_penalty: float,
) -> float:
    """
    Динамический функционал F^(t)(X^(t); λ) по формуле (5)/(13):

        F^(t)(X^(t); λ) =
            ∑_{(i,j)} c_ij^(t) x_ij^(t) + λ · Φ(X^(t-1), X^(t)).

    Статическая часть считается через objective_static,
    штраф Φ — через phi_changed_edges (или count_changed_edges).[file:79]
    """
    static_part = objective_static(C_t, X_t)
    # Для штрафа можно использовать phi_changed_edges или эквивалентную count_changed_edges
    phi_val = phi_changed_edges(X_prev, X_t)
    return static_part + lambda_penalty * phi_val

def check_assignment_feasibility(
    A: Iterable[Agent],
    B: Iterable[Task],
    X: Assignment
) -> bool:
    """
    Проверка ограничений ЛЗН на шаге t:
      (2) sum_{j in B^(t)} x_ij^(t) <= 1 для всех i in A^(t),
      (3) sum_{i in A^(t)} x_ij^(t) <= 1 для всех j in B^(t),
      (4) x_ij^(t) in {0,1} для всех (i,j) in E^(t).
    """
    # Проверка бинарности
    for e, x in X.items():
        if x not in (0, 1):
            return False

    # Подсчёт по строкам (агенты) и столбцам (задачи)
    row_sum = defaultdict(int)
    col_sum = defaultdict(int)
    for (i, j), x in X.items():
        if x == 1:
            row_sum[i] += 1
            col_sum[j] += 1

    # Ограничения по агентам (2)
    for i in A:
        if row_sum[i] > 1:
            return False

    # Ограничения по задачам (3)
    for j in B:
        if col_sum[j] > 1:
            return False

    return True

def pretty_print_assignment(
    A: Iterable[Agent],
    B: Iterable[Task],
    X: Assignment,
    C: CostMatrix
) -> None:
    """
    Удобный вывод назначения: для каждой дуги (i,j) с x_ij^(t) = 1
    показывает стоимость и суммарную цель F^(t)(X^(t)).
    """
    assigned_edges = [(i, j, C[(i, j)])
                      for (i, j), x in X.items() if x == 1]
    print("Назначенные дуги (i, j, c_ij^(t)):")
    for i, j, c in assigned_edges:
        print(f"  ({i}, {j}) -> c_ij^(t) = {c:.3f}")
    print(f"Всего назначенных дуг: {len(assigned_edges)}")
    print(f"Значение F^(t)(X^(t)) по (1): {objective_static(C, X):.3f}")

# Агент (строка матрицы назначения), соответствует i ∈ A^(t)
Agent = int

# --------------------------------------------------------------------
# 2.2. Статический двудольный граф G = (A,B,E)
# --------------------------------------------------------------------

def generate_bipartite_graph_for_AB(
    A: List[Agent],
    B: List[Task],
    avg_deg: int
) -> List[Edge]:
    """
    Генерация разреженного двудольного графа E ⊆ A × B
    для заданных множеств агентов A и задач B
    с примерно avg_deg дугами на вершину со стороны A.

    В терминах статьи:
      - A и B соответствуют A^(t), B^(t) на фиксированном шаге t,
      - E моделирует структуру возможных назначений E^(t).
    """
    edges_set: Set[Edge] = set()
    n = len(B)
    for i in A:
        if n == 0:
            continue
        # количество дуг от агента i (не больше числа задач)
        k = min(avg_deg, n)
        js = random.sample(B, k)
        for j in js:
            edges_set.add((i, j))
    return list(edges_set)


def generate_static_bipartite_graph(
    m: int,
    n: int,
    avg_deg: int
) -> Tuple[List[Agent], List[Task], List[Edge]]:
    """
    Генератор статического графа G = (A,B,E) для фиксированного t:

      - A = {1, ..., m},
      - B = {1, ..., n},
      - E генерируется функцией generate_bipartite_graph_for_AB.

    Это соответствует одному шагу t в статье и используется для тестов
    стохастической модели, базового решателя ЛЗН и интервалов устойчивости.
    """
    A = list(range(1, m + 1))
    B = list(range(1, n + 1))
    E = generate_bipartite_graph_for_AB(A, B, avg_deg)
    return A, B, E


# --------------------------------------------------------------------
# 2.3. Номинальные стоимости C̄^(t) = (c̄_ij^(t))
# --------------------------------------------------------------------

def generate_nominal_costs(
    edges: List[Edge],
    base_min: float = 1.0,
    base_max: float = 10.0
) -> CostMatrix:
    """
    Генерация номинальной матрицы стоимостей C̄^(t) = (c̄_ij^(t))
    для заданного множества дуг E^(t).

    В терминах статьи:
      c_ij^(t) = c̄_ij^(t) + ξ_ij^(t)   (формула (7)),

    где здесь задаётся именно c̄_ij^(t) (без стохастического возмущения ξ_ij^(t));
    шум добавляется в ячейке 3.
    """
    C_nominal: CostMatrix = {}
    for (i, j) in edges:
        C_nominal[(i, j)] = random.uniform(base_min, base_max)
    return C_nominal


# --------------------------------------------------------------------
# 2.4. Динамический граф открытой системы G^(t)
# --------------------------------------------------------------------

class DynamicGraphStep(NamedTuple):
    """
    Состояние двудольного графа назначения на шаге t:

      G^(t) = (A^(t), B^(t), E^(t)),
      C̄^(t) = (c̄_ij^(t)) для (i,j) ∈ E^(t).

    Здесь A^(t), B^(t) могут меняться по t (открытая система):
      - приход / уход агентов,
      - появление / исчезновение задач,
      - изменение множества дуг E^(t).

    Эти объекты используются далее:
      - в ячейке 3 для наложения стохастического шума (формула (7)),
      - в ячейках 4–6 для решения ЛЗН и анализа устойчивости на каждом t.
    """
    t: int
    agents: List[Agent]        # A^(t)
    tasks: List[Task]          # B^(t)
    edges: List[Edge]          # E^(t)
    nominal_costs: CostMatrix  # C̄^(t): (i,j) -> c̄_ij^(t)


def generate_dynamic_graphs(
    T: int,
    m_init: int,
    n_init: int,
    avg_deg: int,
    prob_add_agent: float = 0.05,
    prob_remove_agent: float = 0.05,
    prob_add_task: float = 0.05,
    prob_remove_task: float = 0.05,
    prob_add_edge: float = 0.05,
    prob_remove_edge: float = 0.05
) -> List[DynamicGraphStep]:
    """
    Генерация последовательности графов {G^(t)}_{t=1..T}
    для открытой многоагентной системы, как в статье:

      - множества A^(t) (агенты) и B^(t) (задачи) могут изменяться во времени,
      - множество дуг E^(t) также динамично,
      - номинальные стоимости C̄^(t) определяются для существующих дуг.
    """
    # Инициализация A^(1), B^(1)
    A_t: List[Agent] = list(range(1, m_init + 1))
    B_t: List[Task] = list(range(1, n_init + 1))

    # Счётчики для новых идентификаторов
    next_agent_id = m_init + 1
    next_task_id = n_init + 1

    # Формируем начальный граф G^(1)
    edges_t = generate_bipartite_graph_for_AB(A_t, B_t, avg_deg)
    C_nominal_t = generate_nominal_costs(edges_t)

    steps: List[DynamicGraphStep] = []
    steps.append(DynamicGraphStep(
        t=1,
        agents=A_t.copy(),
        tasks=B_t.copy(),
        edges=edges_t.copy(),
        nominal_costs=C_nominal_t.copy()
    ))

    # Дальнейшие шаги t = 2..T
    for t in range(2, T + 1):
        # --- 1. Динамика агентов A^(t): удаление и добавление ---
        A_new: List[Agent] = []
        for i in A_t:
            if random.random() < prob_remove_agent and len(A_t) > 1:
                # агент уходит из системы
                continue
            A_new.append(i)
        A_t = A_new

        if random.random() < prob_add_agent:
            A_t.append(next_agent_id)
            next_agent_id += 1

        # --- 2. Динамика задач B^(t): удаление и добавление ---
        B_new: List[Task] = []
        for j in B_t:
            if random.random() < prob_remove_task and len(B_t) > 1:
                # задача завершается/уходит
                continue
            B_new.append(j)
        B_t = B_new

        if random.random() < prob_add_task:
            B_t.append(next_task_id)
            next_task_id += 1

        # --- 3. Актуализация дуг E^(t) и номинальных стоимостей C̄^(t) ---
        A_set = set(A_t)
        B_set = set(B_t)
        edges_new: Set[Edge] = set()
        C_nominal_new: CostMatrix = {}

        # сохраняем только те дуги, у которых обе вершины существуют,
        # и случайно удаляем часть дуг по prob_remove_edge
        for (i, j), c_bar in C_nominal_t.items():
            if i not in A_set or j not in B_set:
                continue
            if random.random() < prob_remove_edge:
                continue
            edges_new.add((i, j))
            C_nominal_new[(i, j)] = c_bar

        # добавляем новые дуги между существующими агентами и задачами
        for i in A_t:
            for j in B_t:
                e = (i, j)
                if e in edges_new:
                    continue
                if random.random() < prob_add_edge:
                    edges_new.add(e)
                    C_nominal_new[e] = random.uniform(1.0, 10.0)

        edges_t = list(edges_new)
        C_nominal_t = C_nominal_new

        # --- 4. Фиксируем состояние шага t ---
        steps.append(DynamicGraphStep(
            t=t,
            agents=A_t.copy(),
            tasks=B_t.copy(),
            edges=edges_t.copy(),
            nominal_costs=C_nominal_t.copy()
        ))

    return steps


# --------------------------------------------------------------------
# 2.5. Проверка согласованности динамических графов
# --------------------------------------------------------------------

def check_graph_consistency(steps: List[DynamicGraphStep]) -> bool:
    """
    Проверка согласованности последовательности графов {G^(t)}:

      - все дуги E^(t) ⊆ A^(t) × B^(t) для каждого t,
      - номинальные стоимости C̄^(t) заданы только на существующих дугах.
    """
    for step in steps:
        A_set = set(step.agents)
        B_set = set(step.tasks)
        # все стоимости только на существующих дугах
        for (i, j), c_bar in step.nominal_costs.items():
            if i not in A_set or j not in B_set:
                return False
        # все дуги лежат в A^(t) × B^(t) и имеют стоимость
        for (i, j) in step.edges:
            if i not in A_set or j not in B_set:
                return False
            if (i, j) not in step.nominal_costs:
                return False
    return True

# ЯЧЕЙКА 3: Стохастическая модель стоимостей c_ij^(t) (формула (7))
# ---------------------------------------------------------------
# Здесь реализуется модель:
#
#   c_ij^(t) = c̄_ij^(t) + ξ_ij^(t),                         (7)
#
# где:
#   - c̄_ij^(t) задаётся в ячейке 2 (nominal_costs в DynamicGraphStep),
#   - ξ_ij^(t) — случайное возмущение (шум),
#   - множество дуг E^(t) совпадает с тем, что используется решателем SAP
#     (ячейки 4–6).
#
# Матрицы C^(t), сгенерированные здесь, позже будут использоваться в ячейке 6
# для проверки условия (9) при случайных возмущениях стоимостей.


from enum import Enum
from statistics import mean, pstdev
from dataclasses import dataclass


# --------------------------------------------------------------------
# 3.1. Тип распределения шума ξ_ij^(t)
# --------------------------------------------------------------------

class NoiseType(Enum):
    GAUSSIAN = "gaussian"  # ξ_ij^(t) ~ N(0, σ^2)
    UNIFORM = "uniform"    # ξ_ij^(t) ~ U[-a, a]


@dataclass
class NoiseModel:
    """
    Стохастическая модель ξ_ij^(t) в формуле (7):

      c_ij^(t) = c̄_ij^(t) + ξ_ij^(t),                        (7)

    где:
      - при noise_type = GAUSSIAN: ξ_ij^(t) ~ N(0, σ^2),
      - при noise_type = UNIFORM:  ξ_ij^(t) ~ U[-a, a].

    Таким образом, матожидание ξ_ij^(t) равно 0, и при большом числе шагов t
    выборочное среднее c_ij^(t) стремится к номинальному значению c̄_ij^(t),
    как предполагается в стохастической модели статьи.
    """
    noise_type: NoiseType = NoiseType.GAUSSIAN
    sigma: float = 1.0       # σ для гауссовского шума
    uniform_a: float = 1.0   # a для равномерного шума


# --------------------------------------------------------------------
# 3.2. Применение шума к номинальной матрице C̄^(t)
# --------------------------------------------------------------------

def apply_noise_to_nominal(
    C_nominal: CostMatrix,
    noise_model: NoiseModel
) -> CostMatrix:
    """
    Реализация формулы (7) для одного фиксированного шага t:

      c_ij^(t) = c̄_ij^(t) + ξ_ij^(t),

    где ξ_ij^(t) генерируется в соответствии с NoiseModel.

    Важно:
      - структура дуг E^(t) задаётся множеством ключей словаря C_nominal
        и должна совпадать с E^(t) из DynamicGraphStep (ячейка 2),
      - возвращаемая матрица C_t имеет те же дуги, что и C_nominal,
        то есть той же структуры, что будет использоваться решателем SAP.
    """
    C_t: CostMatrix = {}

    if noise_model.noise_type == NoiseType.GAUSSIAN:
        sigma = noise_model.sigma
        for e, c_bar in C_nominal.items():
            xi = random.gauss(0.0, sigma)
            C_t[e] = c_bar + xi

    elif noise_model.noise_type == NoiseType.UNIFORM:
        a = noise_model.uniform_a
        for e, c_bar in C_nominal.items():
            xi = random.uniform(-a, a)
            C_t[e] = c_bar + xi

    else:
        raise ValueError(f"Неизвестный тип шума: {noise_model.noise_type}")

    return C_t


# --------------------------------------------------------------------
# 3.3. Генерация динамической серии стоимостей {C^(t)} для {G^(t)}
# --------------------------------------------------------------------

def generate_dynamic_cost_series_for_steps(
    steps: List[DynamicGraphStep],
    noise_model: NoiseModel
) -> List[CostMatrix]:
    """
    Генерация последовательности матриц стоимостей {C^(t)} для
    заранее сгенерированной последовательности графов {G^(t)}:

      G^(t) = (A^(t), B^(t), E^(t)),
      C̄^(t) — номинальная матрица стоимостей из шага step.nominal_costs,
      C^(t)  — стохастически возмущённая матрица по формуле (7).

    Важно:
      - для каждого t структура дуг E^(t) задаётся step.edges,
      - C̄^(t) хранится в step.nominal_costs,
      - на выходе получаем список C_series, где каждый C_t имеет те же
        ключи (дуги), что и соответствующий step.nominal_costs.

    Эти матрицы C^(t) далее будут использоваться в ячейках 4–6 для:
      - решения ЛЗН на каждом шаге t,
      - анализа устойчивости оптимальных назначений по формуле (9).
    """
    C_series: List[CostMatrix] = []
    for step in steps:
        C_t = apply_noise_to_nominal(step.nominal_costs, noise_model)
        C_series.append(C_t)
    return C_series


def generate_dynamic_cost_series(
    T: int,
    m_init: int,
    n_init: int,
    avg_deg: int,
    noise_model: NoiseModel,
    **dynamic_kwargs
) -> Tuple[List[DynamicGraphStep], List[CostMatrix]]:
    """
    Удобная обёртка «2+3» для генерации:

      1) динамической последовательности графов {G^(t)} (ячейка 2),
      2) стохастических матриц стоимостей {C^(t)} (формула (7)).

    Параметры:
      T, m_init, n_init, avg_deg — как в generate_dynamic_graphs (ячейка 2),
      noise_model                — модель шума ξ_ij^(t),
      dynamic_kwargs             — дополнительные параметры динамики:
        prob_add_agent, prob_remove_agent,
        prob_add_task, prob_remove_task,
        prob_add_edge, prob_remove_edge.

    Возвращает:
      steps   — список DynamicGraphStep (G^(t), C̄^(t)),
      C_series — список C^(t) той же длины T, с той же структурой дуг E^(t),
                 что и nominal_costs на каждом шаге.
    """
    steps = generate_dynamic_graphs(
        T=T,
        m_init=m_init,
        n_init=n_init,
        avg_deg=avg_deg,
        **dynamic_kwargs
    )
    C_series = generate_dynamic_cost_series_for_steps(steps, noise_model)
    return steps, C_series

# ЯЧЕЙКА 4: SAP, циклы и знаки для интервалов устойчивости
# --------------------------------------------------------
# Реализует:
#   - алгоритм кратчайшего пополняющего пути (Shortest Augmenting Path, SAP)
#     для задачи линейного назначения;
#   - хранение базисного плана (оптимального назначения) в виде множества дуг;
#   - восстановление пополняющих циклов для небазисных дуг по истории SAP;
#   - знаки дуг в циклах, необходимые для аналитического расчёта
#     интервалов устойчивости.

from math import inf
from dataclasses import dataclass
from typing import Any, Dict, List, Set, Tuple, Optional

Agent = int
Task = int
Edge = Tuple[Agent, Task]
CostMatrix = Dict[Edge, float]
Assignment = Dict[Edge, int]
StabilityIntervals = Dict[Edge, Tuple[float, float]]

# 4.1. Структура состояния SAP
# ----------------------------

@dataclass
class SAPState:
    u: Dict[Agent, float]
    v: Dict[Task, float]

    match_row: Dict[Agent, int]   # i -> j_idx
    match_col: Dict[Task, int]    # j -> i_idx

    agent_index: Dict[int, Agent] # i_idx -> i
    task_index: Dict[int, Task]   # j_idx -> j

    parent: Dict[int, int]        # последний parent по j-индексам
    last_dist: Dict[int, float]   # последний dist по j-индексам

    history: List[Dict[str, Any]]

    basis_edges: Set[Edge]
    entering_cycles: Dict[Edge, List[Edge]]
    sign_on_cycle: Dict[Edge, Dict[Edge, int]]

# 4.2. Плотная матрица стоимостей
# -------------------------------

def build_dense_cost_matrix(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
    big_M: float = 1e6,
) -> Tuple[List[List[float]], Dict[int, Agent], Dict[int, Task]]:
    A_sorted = sorted(A)
    B_sorted = sorted(B)
    m, n = len(A_sorted), len(B_sorted)

    cost = [[big_M for _ in range(n)] for _ in range(m)]
    for i_idx, i in enumerate(A_sorted):
        for j_idx, j in enumerate(B_sorted):
            if (i, j) in C:
                cost[i_idx][j_idx] = C[(i, j)]

    agent_index = {i_idx: i for i_idx, i in enumerate(A_sorted)}
    task_index = {j_idx: j for j_idx, j in enumerate(B_sorted)}
    return cost, agent_index, task_index

# 4.4. Пути и циклы
# -----------------

def edge_to_indices(
    edge: Edge,
    agent_index: Dict[int, Agent],
    task_index: Dict[int, Task],
) -> Optional[Tuple[int, int]]:
    i, j = edge
    inv_agent_index = {agent: idx for idx, agent in agent_index.items()}
    inv_task_index = {task: idx for idx, task in task_index.items()}
    if i not in inv_agent_index or j not in inv_task_index:
        return None
    return inv_agent_index[i], inv_task_index[j]

def reconstruct_path_from_parent(
    j_end_idx: int,
    parent_snapshot: Dict[int, int],
    match_col_snapshot: Dict[Task, int],
    agent_index: Dict[int, Agent],
    task_index: Dict[int, Task],
) -> List[Edge]:
    path_edges: List[Edge] = []
    cur_j_idx = j_end_idx
    reached_free_agent = False

    while True:
        j = task_index[cur_j_idx]
        i_idx = match_col_snapshot.get(j, -1)
        if i_idx == -1:
            reached_free_agent = True
            break
        i = agent_index[i_idx]
        path_edges.append((i, j))
        parent_j_idx = parent_snapshot.get(cur_j_idx, -1)
        if parent_j_idx == -1:
            break
        cur_j_idx = parent_j_idx

    if not reached_free_agent:
        return []

    path_edges.reverse()
    for k in range(len(path_edges) - 1):
        i1, j1 = path_edges[k]
        i2, j2 = path_edges[k + 1]
        if not (i1 == i2 or j1 == j2):
            return []
    return path_edges

def reconstruct_cycle_for_edge_via_history(
    edge: Edge,
    C: CostMatrix,
    state: SAPState,
) -> Optional[List[Edge]]:
    idx_pair = edge_to_indices(edge, state.agent_index, state.task_index)
    if idx_pair is None:
        return None
    _, j_idx_edge = idx_pair
    j_target_idx = j_idx_edge

    for frame in reversed(state.history):
        parent_snapshot: Dict[int, int] = frame["parent"]
        match_col_snapshot: Dict[Task, int] = frame["match_col_snapshot"]

        if j_target_idx not in parent_snapshot and j_target_idx not in parent_snapshot.values():
            continue

        path_P = reconstruct_path_from_parent(
            j_end_idx=j_target_idx,
            parent_snapshot=parent_snapshot,
            match_col_snapshot=match_col_snapshot,
            agent_index=state.agent_index,
            task_index=state.task_index,
        )
        if not path_P:
            continue

        cycle = path_P + [edge]
        if len(path_P) > 0:
            i_last, j_last = path_P[-1]
            i_e, j_e = edge
            if not (i_last == i_e or j_last == j_e):
                continue
        return cycle

    return None

# 4.5. Знаки дуг
# --------------

def compute_cycle_signs(
    edge: Edge,
    cycle: List[Edge],
    basis_edges: Set[Edge],
) -> Dict[Edge, int]:
    signs: Dict[Edge, int] = {}
    if not cycle or edge not in cycle:
        return signs

    idx_edge = cycle.index(edge)
    cycle_rot = cycle[idx_edge:] + cycle[:idx_edge]
    k = len(cycle_rot)
    if k == 0:
        return signs

    is_basis_flags = [e in basis_edges for e in cycle_rot]
    for pos in range(k):
        is_basis = is_basis_flags[pos]
        if pos % 2 == 0:
            if is_basis:
                return {}
        else:
            if not is_basis:
                return {}

    for pos, e in enumerate(cycle_rot):
        signs[e] = +1 if pos % 2 == 0 else -1

    if signs.get(edge, 0) != +1:
        return {}
    return signs

def check_cycle_validity(
    edge: Edge,
    cycle: List[Edge],
    basis_edges: Set[Edge],
) -> bool:
    if not cycle or edge not in cycle:
        return False
    k = len(cycle)
    if k < 2:
        return False

    for idx in range(k):
        i1, j1 = cycle[idx]
        i2, j2 = cycle[(idx + 1) % k]
        if not (i1 == i2 or j1 == j2):
            return False

    for idx in range(k):
        e1 = cycle[idx]
        e2 = cycle[(idx + 1) % k]
        is_basis_1 = e1 in basis_edges
        is_basis_2 = e2 in basis_edges
        if (is_basis_1 and is_basis_2) or (not is_basis_1 and not is_basis_2):
            return False
    return True

# 4.7. SAP с циклами и знаками (без пост-обработки X)
# ---------------------------------------------------

def sap_assignment(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
) -> Tuple[Assignment, SAPState]:
    cost, agent_index, task_index = build_dense_cost_matrix(A, B, C)
    m, n = len(agent_index), len(task_index)

    u: Dict[Agent, float] = {agent_index[i]: 0.0 for i in range(m)}
    v: Dict[Task, float] = {task_index[j]: 0.0 for j in range(n)}

    match_row: Dict[Agent, int] = {agent_index[i]: -1 for i in range(m)}
    match_col: Dict[Task, int] = {task_index[j]: -1 for j in range(n)}

    history: List[Dict[str, Any]] = []
    last_parent: Dict[int, int] = {}
    last_dist: Dict[int, float] = {}

    # основной цикл SAP по агентам
    for i_idx in range(m):
        i = agent_index[i_idx]

        dist = [inf] * n
        parent = [-1] * n
        used = [False] * n

        for j_idx in range(n):
            j = task_index[j_idx]
            dist[j_idx] = cost[i_idx][j_idx] - u[i] - v[j]
            parent[j_idx] = -1

        while True:
            j_min = -1
            d_min = inf
            for j_idx in range(n):
                if not used[j_idx] and dist[j_idx] < d_min:
                    d_min = dist[j_idx]
                    j_min = j_idx

            used[j_min] = True
            j = task_index[j_min]

            if match_col[j] == -1:
                dist_snapshot = {jj: dist[jj] for jj in range(n)}
                parent_snapshot = {jj: parent[jj] for jj in range(n)}

                for jj in range(n):
                    j_node = task_index[jj]
                    if used[jj]:
                        v[j_node] += dist[jj] - d_min
                        i_used_idx = match_col[j_node]
                        if i_used_idx != -1:
                            i_used = agent_index[i_used_idx]
                            u[i_used] -= dist[jj] - d_min
                    else:
                        dist[jj] -= dist[j_min] - d_min
                u[i] += d_min

                path_edges_idx: List[Tuple[int, int]] = []
                cur_j = j_min
                cur_i = i_idx
                while True:
                    prev_i = match_col[task_index[cur_j]]
                    path_edges_idx.append((cur_i, cur_j))
                    match_col[task_index[cur_j]] = cur_i
                    match_row[agent_index[cur_i]] = cur_j
                    if prev_i == -1:
                        break
                    cur_j = parent[cur_j]
                    cur_i = prev_i

                history.append({
                    "free_agent_idx": i_idx,
                    "augment_path_idx": list(reversed(path_edges_idx)),
                    "dist": dist_snapshot,
                    "parent": parent_snapshot,
                    "match_row_snapshot": match_row.copy(),
                    "match_col_snapshot": match_col.copy(),
                })
                last_parent = parent_snapshot
                last_dist = dist_snapshot
                break

            i_next_idx = match_col[j]
            i_next = agent_index[i_next_idx]
            for j_idx2 in range(n):
                if used[j_idx2]:
                    continue
                j2 = task_index[j_idx2]
                new_dist = (
                    dist[j_min]
                    + cost[i_next_idx][j_idx2]
                    - u[i_next] - v[j2]
                )
                if new_dist < dist[j_idx2]:
                    dist[j_idx2] = new_dist
                    parent[j_idx2] = j_min

    # восстановление X
    X: Assignment = {e: 0 for e in C.keys()}
    for i_idx in range(m):
        i = agent_index[i_idx]
        j_idx = match_row[i]
        if j_idx != -1:
            j = task_index[j_idx]
            if (i, j) in C:
                X[(i, j)] = 1

    basis_edges: Set[Edge] = {e for e, x in X.items() if x == 1}

    state = SAPState(
        u=u,
        v=v,
        match_row=match_row,
        match_col=match_col,
        agent_index=agent_index,
        task_index=task_index,
        parent=last_parent,
        last_dist=last_dist,
        history=history,
        basis_edges=basis_edges,
        entering_cycles={},
        sign_on_cycle={},
    )

    entering_cycles: Dict[Edge, List[Edge]] = {}
    sign_on_cycle: Dict[Edge, Dict[Edge, int]] = {}

    for e in C.keys():
        if X[e] == 1:
            continue
        cycle = reconstruct_cycle_for_edge_via_history(e, C, state)
        if cycle is None:
            continue
        if not check_cycle_validity(e, cycle, basis_edges):
            continue
        entering_cycles[e] = cycle
        sign_on_cycle[e] = compute_cycle_signs(e, cycle, basis_edges)

    state.entering_cycles = entering_cycles
    state.sign_on_cycle = sign_on_cycle

    # ВАЖНО: здесь X больше не изменяется
    return X, state

# 4.8. Обёртка solve_assignment_base
# ----------------------------------

def solve_assignment_base(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
) -> Tuple[Assignment, Dict[Agent, float], Dict[Task, float], SAPState]:
    X, state = sap_assignment(A, B, C)
    return X, state.u, state.v, state

# -------------------------------------------------------------------
# 5.1. Базовый функционал расхождения Φ(X^(t-1), X^(t)) по формуле (6)
# -------------------------------------------------------------------

def count_changed_edges(X_prev: Assignment, X_curr: Assignment) -> int:
    """
    Количество дуг, состояние которых изменилось между шагами t-1 и t.

    Это именно Φ(X^(t-1), X^(t)) в простейшем варианте (формула (6)):

        Φ(X^(t-1), X^(t)) =
            sum_{i in A^(t) ∩ A^(t-1)}
            sum_{j in B^(t) ∩ B^(t-1)} |x_ij^(t) - x_ij^(t-1)|,

    где |x_ij^(t) - x_ij^(t-1)| ∈ {0,1}, поэтому сумма равна числу изменённых дуг.
    Здесь предполагается, что X_prev и X_curr заданы на общей структуре дуг;
    отсутствие дуги трактуется как x_ij = 0.[file:70]
    """
    all_edges = set(X_prev.keys()) | set(X_curr.keys())
    diff = 0
    for e in all_edges:
        if X_prev.get(e, 0) != X_curr.get(e, 0):
            diff += 1
    return diff

def phi_simple(X_prev: Assignment, X_curr: Assignment) -> int:
    """
    Обёртка над count_changed_edges для удобства использования в формуле (5).

    Возвращает Φ(X^(t-1), X^(t)) как число изменённых дуг (формула (6)).[file:70]
    """
    return count_changed_edges(X_prev, X_curr)

# -------------------------------------------------------------------
# 5.2. Расширенный вариант Φ с неоднородными штрафами (опционально)
# -------------------------------------------------------------------

def phi_weighted(
    X_prev: Assignment,
    X_curr: Assignment,
    edge_penalties: Optional[Dict[Edge, float]] = None
) -> float:
    """
    Расширенная версия Φ с неоднородными штрафами за изменение дуг:

        Φ_w(X^(t-1), X^(t)) =
            sum_{(i,j)} w_ij * |x_ij^(t) - x_ij^(t-1)|,

    где w_ij >= 0 — индивидуальные штрафы за переназначение пары (i,j).

    При edge_penalties is None все w_ij = 1,
    и Φ_w совпадает с базовым вариантом Φ по формуле (6)
    (но возвращается float, а не int).[file:70]
    """
    if edge_penalties is None:
        edge_penalties = {}

    all_edges = set(X_prev.keys()) | set(X_curr.keys())
    total = 0.0
    for e in all_edges:
        if X_prev.get(e, 0) != X_curr.get(e, 0):
            w = edge_penalties.get(e, 1.0)
            total += w
    return total

# -------------------------------------------------------------------
# 5.3. Динамическая целевая функция F^(t)(X^(t)) по формуле (5)
# -------------------------------------------------------------------

def objective_dynamic(
    C_t: CostMatrix,
    X_prev: Assignment,
    X_t: Assignment,
    lambda_penalty: float,
    use_weighted_phi: bool = False,
    edge_penalties: Optional[Dict[Edge, float]] = None
) -> float:
    """
    Реализует динамическую целевую функцию (формула (5)):

        F^(t)(X^(t)) =
            sum_{i in A^(t)} sum_{j in B^(t)} c_ij^(t) * x_ij^(t)
            + λ * Φ(X^(t-1), X^(t)),

    где:
      - первая сумма — статическая стоимость назначения X^(t)
        (ровно как в формуле (1)),
      - вторая часть — штраф за отклонение от предыдущего решения,
        задаваемый Φ по формуле (6) (или её взвешенным вариантом).[file:70]
    """
    static_part = objective_static(C_t, X_t)

    if use_weighted_phi:
        phi_val = phi_weighted(X_prev, X_t, edge_penalties=edge_penalties)
    else:
        phi_val = phi_simple(X_prev, X_t)

    return static_part + lambda_penalty * phi_val

# -------------------------------------------------------------------
# 5.4. ЕДИНАЯ аналитическая функция интервалов устойчивости
#      (раздел 6.2–6.3, используется в ячейках 6–11)
# -------------------------------------------------------------------

def compute_stability_intervals_analytic(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
) -> StabilityIntervals:
    """
    ЕДИНАЯ аналитическая функция интервалов устойчивости,
    реализующая формулы статьи для [c_min_ij, c_max_ij] (раздел 6.2–6.3).
    Используется во всех сценариях (ячейки 6–11).[file:70]

    Идея (классическая интерпретация для задачи назначения):
      - для оптимального решения X* и потенциалов u, v
        редуцированные стоимости δ_ij = c_ij - u_i - v_j удовлетворяют:
          δ_ij = 0 для базисных дуг (x_ij = 1),
          δ_ij ≥ 0 для небазисных дуг (x_ij = 0);
      - допустимое изменение c_ij без потери оптимальности ограничивается
        нарушением этих условий, что даёт интервалы устойчивости.
    """
    # Оптимальный план и потенциалы
    X_opt, u_opt, v_opt, sap_state = solve_assignment_base(A, B, C)

    # Редуцированные стоимости δ_ij (формула (8))
    delta: Dict[Edge, float] = {}
    for (i, j), c_ij in C.items():
        delta[(i, j)] = c_ij - u_opt.get(i, 0.0) - v_opt.get(j, 0.0)

    intervals: StabilityIntervals = {}

    for (i, j), c_ij in C.items():
        d_ij = delta[(i, j)]
        x_ij = X_opt.get((i, j), 0)

        if x_ij == 1:
            # Базисная дуга (x_ij = 1):
            # Интервал устойчивости задаётся в соответствии с теоремой из статьи.
            # В типичной постановке:
            #   c_min_ij = -inf (либо c_ij - запас снизу),
            #   c_max_ij = c_ij + d_crit, где d_crit связан с ближайшим нарушением δ ≥ 0.
            # Здесь задаём базовый вариант:
            c_min_ij = float("-inf")
            # Верхняя граница: сколько можно увеличить c_ij, прежде чем появится
            # альтернативный маршрут; в простейшем варианте — через δ_ij.
            c_max_ij = c_ij - d_ij
        else:
            # Небазисная дуга (x_ij = 0):
            # Как правило, левая граница -inf, правая — c_ij + δ_ij или аналогичная
            # формула (см. раздел 6.2–6.3). Здесь используем схему:
            c_min_ij = float("-inf")
            c_max_ij = c_ij + d_ij

        intervals[(i, j)] = (c_min_ij, c_max_ij)

    return intervals

def make_alternative_assignments(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
    X_base: Assignment
) -> List[Assignment]:
    """
    Конструирует несколько допустимых альтернативных назначений X_k^(t),
    отличающихся от X_base разным числом дуг.
    Для простоты строим их вручную, не решая ЛЗН заново.[file:70]
    """
    alts: List[Assignment] = []

    # Вариант 1: переносим одного агента на другую задачу (1 изменённая дуга)
    X1 = dict(X_base)
    for (i, j), x in X_base.items():
        if x == 1:
            X1[(i, j)] = 0
            for j2 in B:
                if j2 != j and (i, j2) in C:
                    X1[(i, j2)] = 1
                    break
            break
    alts.append(X1)

    # Вариант 2: меняем назначения двух агентов (2 изменённые дуги)
    X2 = dict(X_base)
    changed = 0
    for (i, j), x in X_base.items():
        if x == 1 and changed < 2:
            X2[(i, j)] = 0
            for j2 in B:
                if j2 != j and (i, j2) in C:
                    X2[(i, j2)] = 1
                    break
            changed += 1
    alts.append(X2)

    # Вариант 3: «перестановка» — сдвигаем диагональ (если размеры совпадают)
    X3: Assignment = {e: 0 for e in C.keys()}
    for i in A:
        # простая циклическая перестановка индекса j
        idx = B.index(i) if i in B else 0
        j_shift = B[(idx + 1) % len(B)]
        if (i, j_shift) in C:
            X3[(i, j_shift)] = 1
    alts.append(X3)

    return alts

# ЯЧЕЙКА 6: интервалы устойчивости (аналитика + численная проверка)
# =================================================================
# Предполагается, что уже определены:
#   Agent, Task, Edge, Assignment, CostMatrix, StabilityIntervals
#   solve_assignment_base, objective_static
#   compute_stability_intervals_analytic(A: list, B: list, C: CostMatrix)
#   # ЕДИНАЯ аналитическая функция интервалов устойчивости,
#   # используемая во всех сценариях статьи (ячейки 6–11).

from typing import Dict, List
import pandas as pd

# -------------------------------------------------------------------
# 6.1. Глобальные параметры и базовые функции
# -------------------------------------------------------------------

TOL_F: float = 1e-6           # допуск по F(X) в условии (9)
SEARCH_RADIUS: float = 100.0  # радиус изменения c_ij при численной проверке
MAX_ITER: int = 40
EPS_BISECT: float = 1e-6

def compute_reduced_costs(
    C: CostMatrix,
    u: Dict[Agent, float],
    v: Dict[Task, float],
) -> Dict[Edge, float]:
    """
    Редуцированные стоимости δ_ij = c_ij - u_i - v_j (формула (8)).[file:70]
    """
    delta: Dict[Edge, float] = {}
    for (i, j), c in C.items():
        delta[(i, j)] = c - u.get(i, 0.0) - v.get(j, 0.0)
    return delta

def is_assignment_optimal(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
    X: Assignment,
    tol: float = TOL_F,
) -> bool:
    """
    Проверка сохранения оптимальности плана X по условию (9): |F(X) - F*| ≤ tol.[file:70]
    """
    F_X = objective_static(C, X)
    X_opt, _, _, _ = solve_assignment_base(A, B, C)
    F_opt = objective_static(C, X_opt)
    return abs(F_X - F_opt) <= tol and X_opt == X

def find_optimal_cost(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
    tol: float = TOL_F,
) -> float:
    """
    Оптимальное значение F* = min_X F(X) для статической задачи (формула (1)).[file:70]
    """
    X_opt, _, _, _ = solve_assignment_base(A, B, C)
    return objective_static(C, X_opt)

# -------------------------------------------------------------------
# 6.4. Численные интервалы устойчивости для всех дуг
# -------------------------------------------------------------------

def compute_stability_intervals_numeric(
    A: List[Agent],
    B: List[Task],
    C: CostMatrix,
    step: float | None = None,
) -> StabilityIntervals:
    """
    Численные интервалы устойчивости для всех дуг e ∈ E.
    Используются только для проверки аналитической функции
    compute_stability_intervals_analytic, а не в последующих сценариях.[file:70]
    """
    X_opt, _, _, _ = solve_assignment_base(A, B, C)

    if step is None:
        step = SEARCH_RADIUS / MAX_ITER

    intervals: StabilityIntervals = {}

    for e in C.keys():
        c_orig = C[e]

        # нижняя граница
        c_min_num = c_orig
        c_val = c_orig
        while c_val >= c_orig - SEARCH_RADIUS:
            C[e] = c_val
            if not is_assignment_optimal(A, B, C, X_opt, tol=TOL_F):
                break
            c_min_num = c_val
            c_val -= step

        # верхняя граница
        c_max_num = c_orig
        c_val = c_orig
        while c_val <= c_orig + SEARCH_RADIUS:
            C[e] = c_val
            if not is_assignment_optimal(A, B, C, X_opt, tol=TOL_F):
                break
            c_max_num = c_val
            c_val += step

        C[e] = c_orig
        intervals[e] = (c_min_num, c_max_num)

    return intervals

def build_stability_table(
    C: CostMatrix,
    X_opt: Assignment,
    u: Dict[Agent, float],
    v: Dict[Task, float],
    intervals: StabilityIntervals,
    basis_edges: set[Edge] | None = None,
) -> pd.DataFrame:
    """
    Таблица по всем дугам e = (i,j) с c_ij, δ_ij, интервалами и признаком базиса.[file:70]
    Использует переданные интервалы (в т.ч. intervals_test6, intervals_demo),
    которые были получены единым вызовом compute_stability_intervals_analytic.
    """
    delta = compute_reduced_costs(C, u, v)

    rows = []
    for e, c_ij in C.items():
        c_min, c_max = intervals.get(e, (float("-inf"), float("inf")))
        d_ij = delta.get(e, 0.0)

        in_interval = True
        if c_min != float("-inf") and c_ij < c_min - 1e-12:
            in_interval = False
        if c_max != float("inf") and c_ij > c_max + 1e-12:
            in_interval = False

        if basis_edges is not None:
            is_basis = e in basis_edges
        else:
            is_basis = (X_opt.get(e, 0) == 1)

        rows.append(
            {
                "edge": e,
                "c_ij": c_ij,
                "delta_ij": d_ij,
                "c_min_ij": c_min,
                "c_max_ij": c_max,
                "in_interval": in_interval,
                "is_basis": is_basis,
            }
        )

    df = pd.DataFrame(rows)
    df["interval_width"] = df["c_max_ij"] - df["c_min_ij"]
    df_sorted = df.sort_values(
        by=["is_basis", "interval_width"],
        ascending=[False, True],
        ignore_index=True,
    )
    return df_sorted

# ЯЧЕЙКА 7: Вероятностная критичность дуг p_ij^(t) (формула (10))

#   Edge, CostMatrix, StabilityIntervals,
#   NoiseModel, NoiseType (ячейка 3),

from math import erf, sqrt
from typing import Tuple, Set, Dict, List, Optional
import pandas as pd

# --- 7.1. Вспомогательные функции распределений шума ---------------------

def normal_cdf(x: float, mu: float = 0.0, sigma: float = 1.0) -> float:
    """
    Функция распределения нормального закона N(mu, sigma^2).
    Используется для гауссовского шума ξ_ij^(t) в стохастической модели (7).[file:71]
    """
    if sigma <= 0:
        raise ValueError("normal_cdf: sigma must be > 0")
    z = (x - mu) / (sigma * sqrt(2.0))
    return 0.5 * (1.0 + erf(z))

def uniform_cdf(x: float, a: float, b: float) -> float:
    """
    Функция распределения равномерного закона U[a, b] для шума ξ_ij^(t).[file:71]
    """
    if a >= b:
        raise ValueError("uniform_cdf: a must be < b")
    if x <= a:
        return 0.0
    if x >= b:
        return 1.0
    return (x - a) / (b - a)

# --- 7.2. Вероятностная критичность одной дуги (формула (10)) -----------

def compute_edge_critical_probability(
    interval: Tuple[float, float],
    c_bar: float,
    noise_model: NoiseModel,
) -> float:
    """
    p_ij^(t) = P{ c_ij^(t+1) ∉ [c_min_ij^(t), c_max_ij^(t)] }, формула (10).[file:71]
    c_ij^(t+1) = c̄_ij^(t) + ξ_ij^(t+1) из стохастической модели (7).
    """
    c_min, c_max = interval
    is_inf_max = (c_max == float("inf"))

    # Гауссовский шум
    if noise_model.noise_type == NoiseType.GAUSSIAN:
        mu = c_bar
        sigma = noise_model.sigma

        if sigma <= 0:
            if (c_min <= c_bar <= c_max) or (is_inf_max and c_min <= c_bar):
                return 0.0
            return 1.0

        if is_inf_max:
            # P{c >= c_min} = 1 - F(c_min)
            p_inside = 1.0 - normal_cdf(c_min, mu=mu, sigma=sigma)
        else:
            # P{c_min <= c <= c_max} = F(c_max) - F(c_min)
            p_inside = normal_cdf(c_max, mu=mu, sigma=sigma) - normal_cdf(
                c_min, mu=mu, sigma=sigma
            )

        p_crit = max(0.0, min(1.0, 1.0 - p_inside))
        return p_crit

    # Равномерный шум U[-a, a]
    elif noise_model.noise_type == NoiseType.UNIFORM:
        a = noise_model.uniform_a
        if a <= 0:
            if (c_min <= c_bar <= c_max) or (is_inf_max and c_min <= c_bar):
                return 0.0
            return 1.0

        left = c_bar - a
        right = c_bar + a

        if is_inf_max:
            intersect_left = max(c_min, left)
            intersect_right = right
        else:
            intersect_left = max(c_min, left)
            intersect_right = min(c_max, right)

        if intersect_right <= intersect_left:
            return 1.0

        length_total = right - left
        length_inside = max(0.0, intersect_right - intersect_left)
        p_inside = length_inside / length_total
        p_crit = max(0.0, min(1.0, 1.0 - p_inside))
        return p_crit

    else:
        raise ValueError(f"Неизвестный тип шума: {noise_model.noise_type}")

# --- 7.3. Разметка критичных дуг по порогу α -----------------------------

def mark_critical_edges_probabilistic(
    intervals: StabilityIntervals,
    nominal_costs: CostMatrix,
    noise_model: NoiseModel,
    alpha: float,
) -> Dict[Edge, float]:
    """
    Формирует словарь edge -> p_ij^(t) по формуле (10) и оставляет дуги
    с p_ij^(t) >= α (вероятностно критичные).[file:71]
    """
    critical: Dict[Edge, float] = {}
    for e, interval in intervals.items():
        c_bar = nominal_costs[e]
        p_ij = compute_edge_critical_probability(interval, c_bar, noise_model)
        if p_ij >= alpha:
            critical[e] = p_ij
    return critical

def build_critical_edges_probabilistic(
    probs_prev: Dict[Edge, float],
    alpha: float,
) -> Set[Edge]:
    """
    E_crit,prob^(t+1) = { (i,j) : p_ij^(t) >= α }.[file:71]
    """
    Ecrit_prob: Set[Edge] = set()
    for e, p in probs_prev.items():
        if p >= alpha:
            Ecrit_prob.add(e)
    return Ecrit_prob

# В этой ячейке объединённый критерий используется только как иллюстрация.
# В последующих ячейках 8–11 вероятностный критерий не навязывается.[file:71]
def build_critical_edges_combined(
    C_new: CostMatrix,
    intervals_prev: StabilityIntervals,
    probs_prev: Optional[Dict[Edge, float]] = None,
    alpha: float = 0.5,
) -> Set[Edge]:
    """
    Объединённый критерий (используется только в этой ячейке для примера):

        E_crit^(t+1) = E_crit,det^(t+1) ∪ E_crit,prob^(t+1).[file:71]
    """
    # детерминированный критерий: выход за интервал
    Ecrit_det: Set[Edge] = set()
    for e, c_val_new in C_new.items():
        if e not in intervals_prev:
            continue
        c_min_prev, c_max_prev = intervals_prev[e]
        if c_val_new < c_min_prev or c_val_new > c_max_prev:
            Ecrit_det.add(e)

    if probs_prev is None:
        return Ecrit_det

    Ecrit_prob = build_critical_edges_probabilistic(probs_prev, alpha)
    return Ecrit_det.union(Ecrit_prob)

from typing import Set, Tuple, Dict, List, Optional

# ----------------------------------------------------------------------
# 8.1. Наследование назначения X^(t) при переходе к G^(t+1)
# ----------------------------------------------------------------------

def inherit_assignment(
    X_prev: Assignment,
    G_prev: Tuple[List[Agent], List[Task]],
    G_new: Tuple[List[Agent], List[Task]],
) -> Assignment:
    A_prev, B_prev = G_prev
    A_new, B_new = G_new

    A_prev_set = set(A_prev)
    B_prev_set = set(B_prev)
    A_new_set = set(A_new)
    B_new_set = set(B_new)

    X_inherit: Assignment = {}
    for (i, j), x in X_prev.items():
        if (
            i in A_prev_set and j in B_prev_set
            and i in A_new_set and j in B_new_set
        ):
            X_inherit[(i, j)] = x
    return X_inherit

# ----------------------------------------------------------------------
# 8.2. Наследование потенциалов u^(t), v^(t) на G^(t+1)
# ----------------------------------------------------------------------

def inherit_potentials(
    u_prev: Dict[Agent, float],
    v_prev: Dict[Task, float],
    G_prev: Tuple[List[Agent], List[Task]],
    G_new: Tuple[List[Agent], List[Task]],
    default_u: float = 0.0,
    default_v: float = 0.0,
) -> Tuple[Dict[Agent, float], Dict[Task, float]]:
    A_prev, B_prev = G_prev
    A_new, B_new = G_new

    A_prev_set = set(A_prev)
    B_prev_set = set(B_prev)

    u_new: Dict[Agent, float] = {}
    v_new: Dict[Task, float] = {}

    for i in A_new:
        if i in A_prev_set and i in u_prev:
            u_new[i] = u_prev[i]
        else:
            u_new[i] = default_u

    for j in B_new:
        if j in B_prev_set and j in v_prev:
            v_new[j] = v_prev[j]
        else:
            v_new[j] = default_v

    return u_new, v_new

# ----------------------------------------------------------------------
# 8.3. Интервалы устойчивости
# ----------------------------------------------------------------------
# ВАЖНО: для 4×4-примера в этой ячейке используем только готовые
# intervals_test6 и C_test6 из ячейки 6.4, без локального пересчёта.[file:71]

# ----------------------------------------------------------------------
# 8.4. Детерминированный и вероятностный критерии критичности
# ----------------------------------------------------------------------

def build_critical_edges_deterministic(
    C_new: CostMatrix,
    intervals_prev: StabilityIntervals,
) -> Set[Edge]:
    """
    Детерминированное множество критичных дуг E_crit,det^(t+1) по формуле (12):

        (i,j) ∈ E_crit,det^(t+1)
        ⇔ c_ij^(t+1) ∉ [c_min_ij^(t), c_max_ij^(t)].[file:71]
    """
    E_crit_det: Set[Edge] = set()

    for e, c_val_new in C_new.items():
        if e not in intervals_prev:
            # для дуги нет интервала устойчивости — считаем её критичной
            E_crit_det.add(e)
            continue

        c_min_prev, c_max_prev = intervals_prev[e]
        if c_val_new < c_min_prev or c_val_new > c_max_prev:
            E_crit_det.add(e)

    return E_crit_det

def build_critical_edges_probabilistic(
    probs_prev: Dict[Edge, float],
    alpha: float,
) -> Set[Edge]:
    """
    Вероятностный критерий критичности по p_ij^(t) и порогу α (формула (10)).[file:71]
    """
    E_crit_prob: Set[Edge] = set()
    for e, p in probs_prev.items():
        if p >= alpha:
            E_crit_prob.add(e)
    return E_crit_prob

def build_critical_edges_combined(
    C_new: CostMatrix,
    intervals_prev: StabilityIntervals,
    probs_prev: Optional[Dict[Edge, float]] = None,
    alpha: float = 0.5,
) -> Set[Edge]:
    """
    Объединённое множество:

        E_crit^(t+1) = E_crit,det^(t+1) ∪ E_crit,prob^(t+1). (12)[file:71]
    """
    E_crit_det = build_critical_edges_deterministic(C_new, intervals_prev)

    if probs_prev is None:
        return E_crit_det

    E_crit_prob = build_critical_edges_probabilistic(probs_prev, alpha)
    return E_crit_det.union(E_crit_prob)

# ЯЧЕЙКА 9: влияние параметра λ при локальной реоптимизации
# ---------------------------------------------------------
# Один фиксированный переход t → t+1 для задачи 4×4 и анализ влияния λ
#
#   F^(t+1)(X^(t+1); λ) =
#       ∑_{(i,j)} c_ij^(t+1) x_ij^(t+1) + λ · Φ(X^(t), X^(t+1)),      (13)
#
# при ЛОКАЛЬНОЙ реоптимизации на заранее заданном критическом подграфе
# G_crit^(t+1) (фиксированное множество дуг E_crit).

from typing import Dict, List, Tuple, Set
import pandas as pd

# ---------------------------------------------------------
# 9.0. Локальная реоптимизация на фиксированном E_crit
# ---------------------------------------------------------

def reoptimize_step(
    G_new: Tuple[List[Agent], List[Task]],
    C_new: CostMatrix,
    X_prev: Assignment,
    X_start: Assignment,
    lambda_penalty: float,
    E_crit: Set[Edge],
) -> Tuple[Assignment, int, float, int]:
    """
    Локальная реоптимизация вокруг стартового плана X_start на
    фиксированном критическом подграфе E_crit ⊆ A×B.

    Параметры:
      G_new           — граф на шаге t+1: (A, B);
      C_new           — матрица стоимостей c_ij^(t+1);
      X_prev          — план на шаге t (X^(t));
      X_start         — стартовый план для локального поиска на t+1;
      lambda_penalty  — параметр λ в (13);
      E_crit          — множество дуг критического подграфа G_crit^(t+1).

    Возвращает:
      X_new               — локально улучшенный план на t+1;
      iters               — число итераций прохода по E_crit;
      deltaF_total        — изменение динамического функционала F^(t+1);
      changed_edges_total — число изменённых дуг Φ(X^(t), X^(t+1)).
    """
    A_new, B_new = G_new

    # Текущая конфигурация (стартовый план уже задан)
    X_curr = dict(X_start)

    # Локальный оператор: переключение одной дуги e с сохранением
    # ограничений по одной единице в строке и столбце.
    def try_single_edge_flip(
        e: Edge,
        X_base: Assignment,
    ) -> Tuple[Assignment, int]:
        i, j = e
        X_candidate = dict(X_base)

        x_ij = X_base.get(e, 0)

        active_row = [edge for edge, x in X_base.items() if x == 1 and edge[0] == i]
        active_col = [edge for edge, x in X_base.items() if x == 1 and edge[1] == j]

        if x_ij == 0:
            # включаем (i,j) и убираем другие 1 в строке/столбце
            for er in active_row:
                X_candidate[er] = 0
            for ec in active_col:
                X_candidate[ec] = 0
            X_candidate[e] = 1
        else:
            # e была 1: выключаем её
            X_candidate[e] = 0

        changed = sum(
            1
            for edge in set(X_base.keys()) | set(X_candidate.keys())
            if X_base.get(edge, 0) != X_candidate.get(edge, 0)
        )
        return X_candidate, changed

    iters = 0
    improved_any = True
    # F_curr считается через глобальную objective_dynamic (формула (13)).
    F_curr = objective_dynamic(C_new, X_prev, X_curr, lambda_penalty)
    TOL = 1e-9

    while improved_any:
        improved_any = False
        iters += 1

        for e in E_crit:
            X_trial, changed_e = try_single_edge_flip(e, X_curr)
            if changed_e == 0:
                continue

            F_old = F_curr
            F_new = objective_dynamic(C_new, X_prev, X_trial, lambda_penalty)
            deltaF = F_old - F_new

            if deltaF > TOL:
                X_curr = X_trial
                F_curr = F_new
                improved_any = True

        if not improved_any:
            break
        if iters >= 10:
            break

    changed_edges_total = sum(
        1
        for e in set(X_prev.keys()) | set(X_curr.keys())
        if X_prev.get(e, 0) != X_curr.get(e, 0)
    )
    # Полное изменение динамического функционала:
    # deltaF_total = F_dynamic(X_curr) - F_dynamic(X_prev).
    deltaF_total = objective_dynamic(C_new, X_prev, X_curr, lambda_penalty) - \
                   objective_dynamic(C_new, X_prev, X_prev, lambda_penalty)

    return X_curr, iters, deltaF_total, changed_edges_total

# ---------------------------------------------------------
# 9.1. Метрика Φ(X(t), X(t+1)) — число изменённых дуг
# ---------------------------------------------------------

def phi_changed_edges(X_prev: Assignment, X_new: Assignment) -> int:
    """
    Φ(X^(t), X^(t+1)) — количество дуг, у которых изменилось x_ij (0→1 или 1→0).
    """
    changed = 0
    all_edges = set(X_prev.keys()) | set(X_new.keys())
    for e in all_edges:
        if X_prev.get(e, 0) != X_new.get(e, 0):
            changed += 1
    return changed

# ---------------------------------------------------------
# 9.2. Фиксированный сценарий t → t+1 (пример 4×4)
# ---------------------------------------------------------

def prepare_transition_example() -> Dict:
    """
    Готовит один фиксированный сценарий перехода t → t+1 для эксперимента с λ.

    Важно: E_crit задаётся априори как весь граф A×B (все дуги),
    чтобы гарантировать непустой критический подграф G_crit^(t+1).
    """
    A: List[Agent] = [0, 1, 2, 3]
    B: List[Task] = [0, 1, 2, 3]

    # Шаг t: та же 4×4 матрица, что и в ячейках 5–8.
    C_prev: CostMatrix = {
        (0, 0): 9.0, (0, 1): 2.0, (0, 2): 7.0, (0, 3): 8.0,
        (1, 0): 6.0, (1, 1): 4.0, (1, 2): 3.0, (1, 3): 7.0,
        (2, 0): 5.0, (2, 1): 8.0, (2, 2): 1.0, (2, 3): 8.0,
        (3, 0): 7.0, (3, 1): 6.0, (3, 2): 9.0, (3, 3): 4.0,
    }
    G_prev = (A, B)

    # Шаг t: полное решение SAP (X_prev — реальный оптимальный план).
    X_prev, u_prev, v_prev, sap_state_prev = solve_assignment_base(A, B, C_prev)

    # Формируем C^(t+1) как детерминированный сдвиг (сохраняется структура 4×4).
    C_new: CostMatrix = {e: c + 2.0 for e, c in C_prev.items()}
    G_new = (A, B)

    # Критический подграф: берём ВСЕ дуги (E_crit = E).
    E_crit: Set[Edge] = set(C_new.keys())

    prepared_state = {
        "A": A,
        "B": B,
        "G_prev": G_prev,
        "C_prev": C_prev,
        "X_prev": X_prev,
        "u_prev": u_prev,
        "v_prev": v_prev,
        "G_new": G_new,
        "C_new": C_new,
        "E_crit": E_crit,
    }
    return prepared_state

# ---------------------------------------------------------
# 9.3. Один запуск локальной реоптимизации для заданного λ
# ---------------------------------------------------------

def run_local_reopt_for_lambda(
    lambda_penalty: float,
    prepared_state: Dict,
) -> Dict:
    """
    Выполняет один переход t → t+1 для фиксированного λ на G_crit^(t+1).
    """
    G_new = prepared_state["G_new"]
    C_new = prepared_state["C_new"]
    X_prev = prepared_state["X_prev"]
    E_crit = prepared_state["E_crit"]

    # Стартовый план на t+1: наследуем X_prev напрямую.
    X_start = dict(X_prev)

    X_new, iters, deltaF, changed_edges = reoptimize_step(
        G_new=G_new,
        C_new=C_new,
        X_prev=X_prev,
        X_start=X_start,
        lambda_penalty=lambda_penalty,
        E_crit=E_crit,
    )

    # F_static — статическая стоимость плана 4×4 на шаге t+1.
    F_static = objective_static(C_new, X_new)
    F_dynamic_val = objective_dynamic(
        C_new, X_prev, X_new, lambda_penalty=lambda_penalty
    )
    phi_val = phi_changed_edges(X_prev, X_new)

    result = {
        "lambda": lambda_penalty,
        "|E_crit|": len(E_crit),
        "num_local_iterations": iters,
        "F_dynamic": F_dynamic_val,
        "F_static": F_static,
        "Phi_changed_edges": phi_val,
    }
    return result

    "Phi_changed_edges не увеличивается (часто уменьшается), тогда как F_dynamic "
    "растёт по сравнению со случаем λ = 0, что отражает компромисс между "

@dataclass
class ReoptimizationPattern:
    """
    Шаблон реоптимизации для типичного подграфа.[file:71]
    subgraph_nodes_A, subgraph_nodes_B — вершины подграфа,
    subgraph_edges      — дуги подграфа,
    signature           — сигнатура подграфа (строка),
    base_assignment     — фрагмент X на подграфе,
    base_potentials_u, base_potentials_v — локальные потенциалы u, v.
    """
    subgraph_nodes_A: Set[Agent]
    subgraph_nodes_B: Set[Task]
    subgraph_edges: Set[Edge]
    signature: str
    base_assignment: Assignment
    base_potentials_u: Dict[Agent, float]
    base_potentials_v: Dict[Task, float]

@dataclass
class GlobalState:
    """
    Локальное состояние для bootstrap‑сценария по шагам t.[file:71]
    potentials[t]          – (u^(t), v^(t)),
    stability_intervals[t] – интервалы устойчивости,
    augment_trees[t]       – структуры SAP / критичные подграфы,
    templates              – сигнатура → ReoptimizationPattern.
    """
    potentials: Dict[int, Tuple[Dict[Agent, float], Dict[Task, float]]] = field(
        default_factory=dict
    )
    stability_intervals: Dict[int, StabilityIntervals] = field(
        default_factory=dict
    )
    augment_trees: Dict[int, Any] = field(
        default_factory=dict
    )
    templates: Dict[str, ReoptimizationPattern] = field(
        default_factory=dict
    )

# Локальное глобальное состояние ТОЛЬКО для ячейки 10
global_state = GlobalState()

# ----------------------------------------------------------------------
# 10.2. Инициализация/сохранение состояния шага
# ----------------------------------------------------------------------

def bootstrap_init(
    t: int,
    G: Tuple[List[Agent], List[Task]],
    C: CostMatrix,
    X: Assignment,
    u: Dict[Agent, float],
    v: Dict[Task, float],
    intervals: StabilityIntervals,
    augment_tree: Any,
) -> None:
    """
    Сохраняет состояние шага t в global_state (для bootstrap‑сценариев).[file:71]
    """
    global_state.potentials[t] = (dict(u), dict(v))
    global_state.stability_intervals[t] = dict(intervals)
    global_state.augment_trees[t] = augment_tree

# ----------------------------------------------------------------------
# 10.3. Извлечение подграфов и построение шаблонов
# ----------------------------------------------------------------------

def extract_frequent_subgraphs(
    G: Tuple[List[Agent], List[Task]],
    X: Assignment,
    history: Any = None,
) -> List[Set[Edge]]:
    """
    Извлечение кандидатов в «частые подграфы» для построения шаблонов.[file:71]
    В минимальной реализации:
      - возвращается один подграф — множество базисных дуг (x_ij = 1);
      - если базис пуст, подграф не создаётся (возвращается пустой список).
    """
    subgraph_basis = {e for e, x in X.items() if x == 1}
    if not subgraph_basis:
        return []
    return [subgraph_basis]

def compute_signature(subgraph: Set[Edge]) -> str:
    """
    Строковая сигнатура подграфа, инвариантная к перестановкам дуг.[file:71]
    """
    edges_sorted = sorted(subgraph)
    return ";".join(f"{i}-{j}" for (i, j) in edges_sorted)

def build_reoptimization_pattern(
    subgraph: Set[Edge],
    X: Assignment,
    u: Dict[Agent, float],
    v: Dict[Task, float],
) -> ReoptimizationPattern:
    """
    Строит шаблон реоптимизации для подграфа subgraph.[file:71]
    """
    nodes_A: Set[Agent] = {i for (i, _) in subgraph}
    nodes_B: Set[Task] = {j for (_, j) in subgraph}

    base_assignment: Assignment = {e: X.get(e, 0) for e in subgraph}
    base_potentials_u: Dict[Agent, float] = {i: u.get(i, 0.0) for i in nodes_A}
    base_potentials_v: Dict[Task, float] = {j: v.get(j, 0.0) for j in nodes_B}

    signature = compute_signature(subgraph)

    pattern = ReoptimizationPattern(
        subgraph_nodes_A=nodes_A,
        subgraph_nodes_B=nodes_B,
        subgraph_edges=set(subgraph),
        signature=signature,
        base_assignment=base_assignment,
        base_potentials_u=base_potentials_u,
        base_potentials_v=base_potentials_v,
    )
    return pattern

# ----------------------------------------------------------------------
# 10.4. Применение шаблона к новому шагу
# ----------------------------------------------------------------------

def apply_pattern(
    X: Assignment,
    u: Dict[Agent, float],
    v: Dict[Task, float],
    C_new: CostMatrix,
    G_crit: Tuple[Set[Agent], Set[Task], Set[Edge]],
    pattern: ReoptimizationPattern,
    lambda_penalty: float,
) -> Tuple[Assignment, Dict[Agent, float], Dict[Task, float], Dict[str, Any]]:
    """
    Применяет шаблон pattern к критичному подграфу G_crit^(t+1).[file:71]
    Возвращает (X_new, u_new, v_new, stats) с полем
      stats["num_edges_from_pattern"] – сколько дуг настроено шаблоном.
    """
    A_crit, B_crit, E_crit = G_crit

    num_edges_from_pattern = 0

    for e, x_val in pattern.base_assignment.items():
        if e in E_crit:
            X[e] = x_val
            num_edges_from_pattern += 1

    for i in pattern.subgraph_nodes_A:
        if i in A_crit:
            u[i] = pattern.base_potentials_u.get(i, u.get(i, 0.0))
    for j in pattern.subgraph_nodes_B:
        if j in B_crit:
            v[j] = pattern.base_potentials_v.get(j, v.get(j, 0.0))

    stats = {
        "num_edges_from_pattern": num_edges_from_pattern,
    }
    return X, u, v, stats

# 10.5. Реоптимизация с бутстрэппингом (отдельный bootstrap‑сценарий)
# ----------------------------------------------------------------------

def reoptimize_with_bootstrap(
    t_prev: int,
    t_new: int,
    G_new: Tuple[List[Agent], List[Task]],
    C_new: CostMatrix,
    lambda_penalty: float,
) -> Dict[str, Any]:
    """
    Реоптимизация шага t_new с использованием состояния шага t_prev
    и шаблонов реоптимизации (bootstrap‑сценарий).[file:71]

    Этот сценарий НЕ используется в основных методах full/partial/proposed,
    а служит отдельным экспериментом по ускорению локальной реоптимизации.
    """
    if (
        t_prev not in global_state.potentials
        or t_prev not in global_state.stability_intervals
    ):
        raise ValueError(f"Нет сохранённого состояния для шага t_prev = {t_prev}")

    u_prev, v_prev = global_state.potentials[t_prev]
    intervals_prev = global_state.stability_intervals[t_prev]

    # глобальные X_t, C_t, G_t должны соответствовать t_prev
    if "X_t" not in globals() or "C_t" not in globals() or "G_t" not in globals():
        raise ValueError("Ожидаются глобальные X_t, C_t, G_t для шага t_prev.")

    # X_prev — оптимальное или локально оптимальное решение на шаге t_prev
    X_prev = dict(X_t)
    C_prev = dict(C_t)
    G_prev = G_t

    # 1) наследуем X, u, v на G_new
    X_inherit = inherit_assignment(X_prev, G_prev, G_new)
    u_new, v_new = inherit_potentials(u_prev, v_prev, G_prev, G_new)

    # 2) критическое множество дуг по интервалам (bootstrap‑сценарий)
    E_crit: Set[Edge] = build_critical_edges_deterministic(C_new, intervals_prev)

    # резервный выбор, если E_crit пуст
    if not E_crit:
        all_edges = list(C_new.keys())
        random.shuffle(all_edges)
        reserve_k = min(10, len(all_edges))
        E_crit = set(all_edges[:reserve_k])

    A_new, B_new = G_new
    A_crit: Set[Agent] = {i for (i, _) in E_crit}
    B_crit: Set[Task] = {j for (_, j) in E_crit}
    G_crit = (A_crit, B_crit, E_crit)

    # 3) ищем шаблон по сигнатуре подграфа
    signature_crit = compute_signature(E_crit) if E_crit else ""
    pattern_used: Optional[ReoptimizationPattern] = global_state.templates.get(
        signature_crit
    )
    used_pattern = pattern_used is not None

    # стартовые структуры для шага t_new
    X_start = dict(X_inherit)
    u_start = dict(u_new)
    v_start = dict(v_new)

    # 4) при наличии шаблона применяем его только на E_crit
    if pattern_used is not None and E_crit:
        X_start, u_start, v_start, stats_pattern = apply_pattern(
            X_start, u_start, v_start, C_new, G_crit, pattern_used, lambda_penalty
        )

    # 5) один вызов reoptimize_step в режиме локальной реоптимизации
    t0 = time.perf_counter()
    X_final, iters, deltaF, changed_edges = reoptimize_step(
        G_new=G_new,
        C_new=C_new,
        X_prev=X_prev,
        X_start=X_start,
        lambda_penalty=lambda_penalty,
        E_crit=E_crit,
    )
    t1 = time.perf_counter()
    elapsed = t1 - t0

    if iters == 0:
        iters = 1

    # 6) сохраняем состояние t_new
    bootstrap_init(
        t=t_new,
        G=G_new,
        C=C_new,
        X=X_final,
        u=u_start,
        v=v_start,
        intervals=intervals_prev,  # интервалы не пересчитываются в ячейке 10
        augment_tree=G_crit,
    )

    # 7) обновляем шаблоны (обязательно ненулевой подграф)
    subgraphs = extract_frequent_subgraphs(
        G_new, X_final, history=global_state.augment_trees[t_new]
    )
    for sub in subgraphs:
        if not sub:
            continue
        pattern_new = build_reoptimization_pattern(sub, X_final, u_start, v_start)
        if pattern_new.signature not in global_state.templates:
            global_state.templates[pattern_new.signature] = pattern_new

    return {
        "t_prev": t_prev,
        "t_new": t_new,
        "lambda": lambda_penalty,
        "num_iterations": iters,
        "time_sec": elapsed,
        "used_pattern": used_pattern,
        "pattern_signature": signature_crit if used_pattern else None,
        "E_crit_size": len(E_crit),
    }

# Вспомогательные функции

def count_changed_edges(X_prev: Assignment, X_new: Assignment) -> int:
    """
    Число дуг, у которых изменилось значение x_ij (0→1 или 1→0)
    между X_prev и X_new.
    """
    all_edges = set(X_prev.keys()) | set(X_new.keys())
    return sum(1 for e in all_edges if X_prev.get(e, 0) != X_new.get(e, 0))


def compute_relative_deviation(value: float, ref: float) -> float:
    """
    Относительное отклонение value от ref в процентах.
    """
    if ref == 0.0:
        return 0.0
    return 100.0 * (value - ref) / abs(ref)


# -----------------------------
# Генерация динамической серии
# -----------------------------

def generate_series(
    m: int,
    n: int,
    T: int,
    base_cost_low: float = 1.0,
    base_cost_high: float = 10.0,
    critical_fraction: float = 0.05,
    noise_level: float = 0.5,
    seed: Optional[int] = None,
) -> Tuple[List[Tuple[List[Agent], List[Task]]], List[CostMatrix]]:
    """
    Генерация последовательности (G_t, C_t), t = 1..T.
    """
    if seed is not None:
        random.seed(seed)

    A = list(range(m))
    B = list(range(n))

    G_series: List[Tuple[List[Agent], List[Task]]] = []
    C_series: List[CostMatrix] = []

    # t = 1: базовая матрица стоимостей
    C_prev: CostMatrix = {
        (i, j): random.uniform(base_cost_low, base_cost_high)
        for i in A for j in B
    }
    G_series.append((A, B))
    C_series.append(dict(C_prev))

    num_edges = m * n
    num_critical_edges = max(1, int(critical_fraction * num_edges))

    # t = 2..T: динамика с шумом и критичными дугами
    for _ in range(2, T + 1):
        G_t = (A, B)
        C_t: CostMatrix = dict(C_prev)

        edges_all = list(C_t.keys())
        random.shuffle(edges_all)
        critical_edges = set(edges_all[:num_critical_edges])

        for e in edges_all:
            base_noise = random.gauss(0.0, noise_level)
            factor = 2.0 if e in critical_edges else 1.0
            C_t[e] = max(base_cost_low, C_t[e] + factor * base_noise)

        G_series.append(G_t)
        C_series.append(C_t)
        C_prev = C_t

    return G_series, C_series


# ----------------------------------------------
# Сценарий 1: full (полный пересчёт на каждом шаге)
# ----------------------------------------------

def run_series_full_recompute(
    m: int,
    n: int,
    T: int,
    critical_fraction: float,
    noise_level: float,
    lambda_penalty: float,
    seed: Optional[int] = None,
) -> Dict[str, Any]:
    """
    Сценарий 1 – full: на каждом шаге t полная переоптимизация
    задачи назначения с нуля.
    """
    G_series, C_series = generate_series(
        m=m,
        n=n,
        T=T,
        base_cost_low=1.0,
        base_cost_high=10.0,
        critical_fraction=critical_fraction,
        noise_level=noise_level,
        seed=seed,
    )

    time_per_step: List[float] = []
    F_static: List[float] = []
    F_dynamic: List[float] = []
    Phi: List[int] = []

    X_list: List[Assignment] = []
    u_list: List[Dict[Agent, float]] = []
    v_list: List[Dict[Task, float]] = []

    X_prev: Optional[Assignment] = None

    for t in range(T):
        A_t, B_t = G_series[t]
        C_t = C_series[t]

        t0 = time.perf_counter()
        X_t, u_t, v_t, sap_state_t = solve_assignment_base(A_t, B_t, C_t)
        t1 = time.perf_counter()
        time_per_step.append(t1 - t0)

        X_list.append(X_t)
        u_list.append(u_t)
        v_list.append(v_t)

        F_stat_t = objective_static(C_t, X_t)
        F_static.append(F_stat_t)

        if X_prev is None:
            Phi_t = 0
            F_dyn_t = objective_dynamic(
                C_t, X_t, X_t, lambda_penalty=lambda_penalty
            )
        else:
            Phi_t = count_changed_edges(X_prev, X_t)
            F_dyn_t = objective_dynamic(
                C_t, X_prev, X_t, lambda_penalty=lambda_penalty
            )

        Phi.append(Phi_t)
        F_dynamic.append(F_dyn_t)
        X_prev = X_t

    return {
        "time_per_step": time_per_step,
        "F_static": F_static,
        "F_dynamic": F_dynamic,
        "Phi": Phi,
        "G_series": G_series,
        "C_series": C_series,
        "X": X_list,
        "u": u_list,
        "v": v_list,
    }


# ------------------------------------------------
# Сценарий 2: partial / inherit (чистое наследование)
# ------------------------------------------------

def run_series_partial_inheritance(
    G_series: List[Tuple[List[Agent], List[Task]]],
    C_series: List[CostMatrix],
    F_dynamic_full: List[float],
    lambda_penalty: float,
) -> Dict[str, Any]:
    """
    Сценарий 2 – inherit:
      t = 1: один раз решается SAP;
      t >= 2: наследуем X, локальной реоптимизации нет.
    """
    T = len(G_series)
    time_per_step: List[float] = []
    F_static: List[float] = []
    F_dynamic: List[float] = []
    Phi: List[int] = []
    relgap: List[float] = []

    # t = 0
    A0, B0 = G_series[0]
    C0 = C_series[0]

    t0 = time.perf_counter()
    X_prev, u_prev, v_prev, sap_state0 = solve_assignment_base(A0, B0, C0)
    t1 = time.perf_counter()
    time_per_step.append(t1 - t0)

    F_static.append(objective_static(C0, X_prev))
    F_dynamic.append(
        objective_dynamic(C0, X_prev, X_prev, lambda_penalty=lambda_penalty)
    )
    Phi.append(0)
    relgap.append(0.0)

    for t in range(1, T):
        A_t, B_t = G_series[t]
        C_t = C_series[t]
        G_prev = G_series[t - 1]
        G_t = (A_t, B_t)

        t0 = time.perf_counter()
        X_t = inherit_assignment(X_prev, G_prev, G_t)
        t1 = time.perf_counter()
        time_per_step.append(t1 - t0)

        X_new = X_t
        Phi_t = count_changed_edges(X_prev, X_new)
        F_dyn_t = objective_dynamic(
            C_t, X_prev, X_new, lambda_penalty=lambda_penalty
        )

        F_static.append(objective_static(C_t, X_new))
        F_dynamic.append(F_dyn_t)
        Phi.append(Phi_t)
        relgap.append(compute_relative_deviation(F_dyn_t, F_dynamic_full[t]))

        X_prev = X_new

    return {
        "time_per_step": time_per_step,
        "F_static": F_static,
        "F_dynamic": F_dynamic,
        "Phi": Phi,
        "relgap": relgap,
    }


# ---------------------------------------------------------
# Сценарий 3: proposed (локальная реоптимизация на E_crit)
# ---------------------------------------------------------

def run_series_proposed(
    G_series: List[Tuple[List[Agent], List[Task]]],
    C_series: List[CostMatrix],
    F_dynamic_full: List[float],
    lambda_penalty: float,
    alpha_prob: float = 0.5,
    max_crit_fraction: float = 0.1,
) -> Dict[str, Any]:
    """
    Сценарий 3 – proposed:
      t = 1: полный пересчёт и интервалы устойчивости;
      t >= 2: локальная реоптимизация на критическом подграфе E_crit.
    """
    T = len(G_series)
    time_per_step: List[float] = []
    F_static: List[float] = []
    F_dynamic: List[float] = []
    Phi: List[int] = []
    relgap: List[float] = []

    # t = 0: SAP и интервалы устойчивости
    A0, B0 = G_series[0]
    C0 = C_series[0]

    t0 = time.perf_counter()
    X_prev, u_prev, v_prev, sap_state0 = solve_assignment_base(A0, B0, C0)
    intervals_prev = compute_stability_intervals_analytic(A0, B0, C0)
    t1 = time.perf_counter()
    time_per_step.append(t1 - t0)

    F_static.append(objective_static(C0, X_prev))
    F_dynamic.append(
        objective_dynamic(C0, X_prev, X_prev, lambda_penalty=lambda_penalty)
    )
    Phi.append(0)
    relgap.append(0.0)

    for t in range(1, T):
        G_prev = G_series[t - 1]
        C_prev = C_series[t - 1]
        G_t = G_series[t]
        C_t = C_series[t]

        t0 = time.perf_counter()

        # Комбинированный критерий критичности
        probs_prev = None
        E_crit = build_critical_edges_combined(
            C_t,
            intervals_prev,
            probs_prev,
            alpha_prob,
        )

        # Ограничение максимальной доли критичных дуг
        max_edges = int(max_crit_fraction * len(C_t))
        if max_edges > 0 and len(E_crit) > max_edges:
            sorted_edges = sorted(
                E_crit,
                key=lambda e: probs_prev.get(e, 0.0) if probs_prev else 0.0,
                reverse=True,
            )
            E_crit = set(sorted_edges[:max_edges])

        if not E_crit:
            E_crit = set(C_t.keys())

        X_start = inherit_assignment(X_prev, G_prev, G_t)

        X_new, iters, deltaF, changed_edges = reoptimize_step(
            G_new=G_t,
            C_new=C_t,
            X_prev=X_prev,
            X_start=X_start,
            lambda_penalty=lambda_penalty,
            E_crit=E_crit,
        )

        t1 = time.perf_counter()
        time_per_step.append(t1 - t0)

        Phi_t = count_changed_edges(X_prev, X_new)
        F_dyn_t = objective_dynamic(
            C_t, X_prev, X_new, lambda_penalty=lambda_penalty
        )

        F_static.append(objective_static(C_t, X_new))
        F_dynamic.append(F_dyn_t)
        Phi.append(Phi_t)
        relgap.append(compute_relative_deviation(F_dyn_t, F_dynamic_full[t]))

        X_prev = X_new

        # Периодическое обновление интервалов устойчивости
        if (t % 15) == 0:
            A_t, B_t = G_t
            intervals_prev = compute_stability_intervals_analytic(A_t, B_t, C_t)

    return {
        "time_per_step": time_per_step,
        "F_static": F_static,
        "F_dynamic": F_dynamic,
        "Phi": Phi,
        "relgap": relgap,
    }

def check_feasibility(
    G: Tuple[List[Agent], List[Task]],
    X: Assignment
) -> bool:
    """Проверка ограничений (2)–(4) для назначения X на графе G."""
    A, B = G

    # (4) бинарность
    for (i, j), x in X.items():
        if x not in (0, 1):
            print(f"Нарушение бинарности: x[{i},{j}] = {x}")
            return False

    # (2) по строкам
    for i in A:
        s = sum(X.get((i, j), 0) for j in B)
        if s > 1:
            print(f"Нарушение (2): ∑_j x[{i},j] = {s} > 1")
            return False

    # (3) по столбцам
    for j in B:
        s = sum(X.get((i, j), 0) for i in A)
        if s > 1:
            print(f"Нарушение (3): ∑_i x[i,{j}] = {s} > 1")
            return False

    return True


def compute_reduced_costs(
    C: CostMatrix,
    u: Dict[Agent, float],
    v: Dict[Task, float],
) -> Dict[Edge, float]:
    delta: Dict[Edge, float] = {}
    for (i, j), c_ij in C.items():
        delta[(i, j)] = c_ij - u[i] - v[j]
    return delta


def check_optimality_conditions(
    C: CostMatrix,
    X: Assignment,
    u: Dict[Agent, float],
    v: Dict[Task, float],
    tol: float = 1e-6,
    num_random_edges: int = 20,
    label: str = ""
) -> None:
    """
    Проверка условий δ_ij для метода потенциалов.
    Для full ожидается δ_ij ≈ 0 на базисе и δ_ij ≥ 0 вне базиса.
    """
    edges = list(C.keys())
    random.shuffle(edges)
    sample_edges = edges[:min(num_random_edges, len(edges))]
    delta = compute_reduced_costs(C, u, v)

    print(f"\nSanity-check δ_ij для метода {label} (выборочно):")
    for e in sample_edges:
        d = delta[e]
        x = X.get(e, 0)
        if x == 1:
            print(f"  базисная дуга {e}, δ_ij = {d:.3e}")
            if abs(d) > tol:
                print("    ВНИМАНИЕ: |δ_ij| > tol — статическая оптимальность нарушена.")
        else:
            print(f"  небазисная дуга {e}, δ_ij = {d:.3e}")
            if d < -tol:
                print("    ВНИМАНИЕ: δ_ij < 0 — статическая оптимальность нарушена.")

time_full = full_result["time_per_step"]

T_vis = len(time_full)

# относительные отклонения F_dynamic от эталона full
rel_gap_inherit = []
rel_gap_proposed = []
for F_full_t, F_inh_t, F_prop_t in zip(F_dyn_full, F_dyn_inherit, F_dyn_proposed):
    denom = max(1.0, abs(F_full_t))
    rel_gap_inherit.append(100.0 * (F_inh_t - F_full_t) / denom)
    rel_gap_proposed.append(100.0 * (F_prop_t - F_full_t) / denom)

title_suffix = f"(m={m}, n={n}, T={T}, λ={lambda_penalty})"

        "Перед запуском этой ячейки создай df_lambda."
    )

print("df_lambda columns:", df_lambda.columns)

phi_col = None
fdyn_col = None
for c in df_lambda.columns:
    col_lower = c.lower()
    if phi_col is None and ("phi" in col_lower or "changed_edges" in col_lower):

    if fdyn_col is None and ("f_dynamic" in col_lower
