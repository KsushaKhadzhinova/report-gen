% ПРИЛОЖЕНИЕ Б: Листинг программного кода (Рекомендуемое)
\apptitle{Б}{рекомендуемое}{ЛИСТИНГ ПРОГРАММНОГО ОБЕСПЕЧЕНИЯ МИКРОКОНТРОЛЛЕРА}
\begin{verbatim}
#include <WiFi.h>
#include <PubSubClient.h>
#include "SHT31.h"

void setup() {
    Serial.begin(115200);
    setup_wifi();
    client.setServer(mqtt_server, 1883);
}

void loop() {
    float t = sht31.readTemperature();
    float h = sht31.readHumidity();
    publishData(t, h);
    ESP.deepSleep(600e6); // Сон на 10 минут
}
\end{verbatim}
